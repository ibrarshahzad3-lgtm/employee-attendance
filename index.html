const SHEET_NAME = "Attendance"; const EMPLOYEE_SHEET = "Employees"; // ===================================================== // COMPANY LOCATION // ===================================================== const COMPANY_LAT = 24.978665; const COMPANY_LNG = 55.198354; // Employee must be within this distance const ALLOWED_RADIUS_METERS = 200; // ===================================================== // WEB APP // ===================================================== function doGet(e) { if ( e && e.parameter && e.parameter.api === "1" ) { return handleAPI(e); } return HtmlService .createHtmlOutput( "

Employee Attendance API is running

" ) .setTitle("Employee Attendance") .setXFrameOptionsMode( HtmlService.XFrameOptionsMode.ALLOWALL ); } // ===================================================== // API HANDLER // ===================================================== function handleAPI(e) { const callback = cleanCallback( e && e.parameter ? e.parameter.callback : "" ); try { const action = String( e.parameter.action || "" ).trim(); const pin = String( e.parameter.pin || "" ).trim(); // ================================================= // PIN REQUIRED // ================================================= if (!pin) { throw new Error( "PIN is required." ); } let result; // ================================================= // VERIFY EMPLOYEE // ================================================= if ( action === "verify" ) { result = verifyEmployeeAPI( pin ); } // ================================================= // ATTENDANCE // ================================================= else if ( action === "attendance" ) { const attendanceAction = String( e.parameter.attendanceAction || "" ).trim(); const latitude = Number( e.parameter.latitude ); const longitude = Number( e.parameter.longitude ); result = markAttendanceAPI( pin, attendanceAction, latitude, longitude ); } // ================================================= // INVALID API ACTION // ================================================= else { throw new Error( "Invalid API action." ); } // ================================================= // SUCCESS RESPONSE // ================================================= return jsonResponse( { success: true, data: result }, callback ); } // =================================================== // ERROR // =================================================== catch (error) { return jsonResponse( { success: false, error: error && error.message ? error.message : "Server error." }, callback ); } } // ===================================================== // JSONP SECURITY // ===================================================== function cleanCallback( callback ) { if (!callback) { return ""; } callback = String(callback); // IMPORTANT: // Correct JSONP callback validation if ( !/^[a-zA-Z_$][0-9a-zA-Z_$]*$/ .test(callback) ) { return ""; } return callback; } // ===================================================== // JSON RESPONSE // ===================================================== function jsonResponse( data, callback ) { const json = JSON.stringify( data ); // ================================================= // JSONP RESPONSE // ================================================= if (callback) { return ContentService .createTextOutput( callback + "(" + json + ");" ) .setMimeType( ContentService.MimeType.JAVASCRIPT ); } // ================================================= // NORMAL JSON RESPONSE // ================================================= return ContentService .createTextOutput( json ) .setMimeType( ContentService.MimeType.JSON ); } // ===================================================== // FIND EMPLOYEE BY PIN // ===================================================== function findEmployeeByPIN( pin ) { const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName( EMPLOYEE_SHEET ); if (!sheet) { throw new Error( "Employees sheet not found." ); } const lastRow = sheet.getLastRow(); if ( lastRow < 2 ) { throw new Error( "No employees found." ); } const data = sheet .getRange( 2, 1, lastRow - 1, 3 ) .getDisplayValues(); for ( let i = 0; i < data.length; i++ ) { const name = String( data[i][0] ).trim(); const employeePIN = String( data[i][1] ).trim(); const weeklyOff = String( data[i][2] ).trim(); if ( employeePIN === pin ) { return { name: name, weeklyOff: weeklyOff }; } } throw new Error( "Invalid PIN." ); } // ===================================================== // VERIFY EMPLOYEE // ===================================================== function verifyEmployeeAPI( pin ) { const employee = findEmployeeByPIN( pin ); return { name: employee.name, weeklyOff: employee.weeklyOff, status: getCurrentStatus( employee.name ) }; } // ===================================================== // GET CURRENT STATUS // ===================================================== function getCurrentStatus( name ) { const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName( SHEET_NAME ); if (!sheet) { throw new Error( "Attendance sheet not found." ); } const lastRow = sheet.getLastRow(); if ( lastRow < 2 ) { return "Logout"; } const data = sheet .getRange( 2, 1, lastRow - 1, 6 ) .getDisplayValues(); // Search latest record first for ( let i = data.length - 1; i >= 0; i-- ) { const employeeName = String( data[i][2] ).trim(); const loginStatus = String( data[i][3] ).trim(); const logoutStatus = String( data[i][4] ).trim(); if ( employeeName .toLowerCase() === name .toLowerCase() ) { if ( loginStatus === "Login" && logoutStatus !== "Logout" ) { return "Login"; } return "Logout"; } } return "Logout"; } // ===================================================== // DISTANCE CALCULATION // ===================================================== function getDistanceInMeters( lat1, lon1, lat2, lon2 ) { const R = 6371000; const dLat = ( lat2 - lat1 ) * Math.PI / 180; const dLon = ( lon2 - lon1 ) * Math.PI / 180; const a = Math.sin( dLat / 2 ) * Math.sin( dLat / 2 ) + Math.cos( lat1 * Math.PI / 180 ) * Math.cos( lat2 * Math.PI / 180 ) * Math.sin( dLon / 2 ) * Math.sin( dLon / 2 ); const c = 2 * Math.atan2( Math.sqrt(a), Math.sqrt( 1 - a ) ); return R * c; } // ===================================================== // CHECK COMPANY LOCATION // ===================================================== function checkCompanyLocation( latitude, longitude ) { // ================================================= // CHECK NUMBER // ================================================= if ( !Number.isFinite( latitude ) || !Number.isFinite( longitude ) ) { throw new Error( "Unable to get your location. Please allow Location permission." ); } // ================================================= // CHECK GPS RANGE // ================================================= if ( latitude < -90 || latitude > 90 || longitude < -180 || longitude > 180 ) { throw new Error( "Invalid GPS coordinates." ); } // ================================================= // CALCULATE DISTANCE // ================================================= const distance = getDistanceInMeters( latitude, longitude, COMPANY_LAT, COMPANY_LNG ); // ================================================= // CHECK 200 METERS // ================================================= if ( distance > ALLOWED_RADIUS_METERS ) { throw new Error( "You are outside the company area. " + "Login and Logout are allowed only inside the company location." ); } return { allowed: true, distance: Math.round( distance ) }; } // ===================================================== // LOGIN / LOGOUT // ===================================================== function markAttendanceAPI( pin, action, latitude, longitude ) { // ================================================= // CHECK ACTION // ================================================= if ( action !== "Login" && action !== "Logout" ) { throw new Error( "Invalid action." ); } // ================================================= // LOCATION CHECK // ================================================= const location = checkCompanyLocation( latitude, longitude ); // ================================================= // LOCK // ================================================= const lock = LockService .getScriptLock(); lock.waitLock( 10000 ); try { // ================================================= // FIND EMPLOYEE // ================================================= const employee = findEmployeeByPIN( pin ); const employeeName = employee.name; const weeklyOff = employee.weeklyOff; // ================================================= // ATTENDANCE SHEET // ================================================= const attendanceSheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName( SHEET_NAME ); if (!attendanceSheet) { throw new Error( "Attendance sheet not found." ); } const now = new Date(); // ================================================= // LOGIN // ================================================= if ( action === "Login" ) { const lastRow = attendanceSheet .getLastRow(); if ( lastRow >= 2 ) { const data = attendanceSheet .getRange( 2, 1, lastRow - 1, 6 ) .getDisplayValues(); // Search latest record for ( let i = data.length - 1; i >= 0; i-- ) { const name = String( data[i][2] ).trim(); const loginStatus = String( data[i][3] ).trim(); const logoutStatus = String( data[i][4] ).trim(); if ( name .toLowerCase() === employeeName .toLowerCase() && loginStatus === "Login" && logoutStatus !== "Logout" ) { throw new Error( employeeName + " is already logged in." ); } } } // ================================================= // ADD LOGIN RECORD // ================================================= attendanceSheet.appendRow( [ now, now, employeeName, "Login", "", "" ] ); return { name: employeeName, status: "Login", weeklyOff: weeklyOff, distance: location.distance, time: formatTime( now ) }; } // ================================================= // LOGOUT // ================================================= if ( action === "Logout" ) { const lastRow = attendanceSheet .getLastRow(); if ( lastRow < 2 ) { throw new Error( employeeName + " has no login record." ); } const data = attendanceSheet .getRange( 2, 1, lastRow - 1, 6 ) .getDisplayValues(); let loginRow = -1; // Search latest open login for ( let i = data.length - 1; i >= 0; i-- ) { const name = String( data[i][2] ).trim(); const loginStatus = String( data[i][3] ).trim(); const logoutStatus = String( data[i][4] ).trim(); if ( name .toLowerCase() === employeeName .toLowerCase() && loginStatus === "Login" && logoutStatus !== "Logout" ) { loginRow = i + 2; break; } } // ================================================= // NO LOGIN FOUND // ================================================= if ( loginRow === -1 ) { throw new Error( employeeName + " is not currently logged in." ); } // ================================================= // UPDATE LOGOUT STATUS // ================================================= attendanceSheet .getRange( loginRow, 5 ) .setValue( "Logout" ); // ================================================= // UPDATE LOGOUT TIME // ================================================= attendanceSheet .getRange( loginRow, 6 ) .setValue( now ); return { name: employeeName, status: "Logout", weeklyOff: weeklyOff, distance: location.distance, time: formatTime( now ) }; } throw new Error( "Invalid attendance action." ); } finally { lock.releaseLock(); } } // ===================================================== // FORMAT TIME // ===================================================== function formatTime( date ) { return Utilities.formatDate( date, Session.getScriptTimeZone(), "HH:mm:ss" );
